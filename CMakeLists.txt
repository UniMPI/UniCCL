# UniCCL - unified NCCL/RCCL collective-communication wrapper.
# M2 skeleton. Builds with a plain C compiler on the host; nothing in the
# build depends on nccl.h / rccl.h / CUDA / ROCm headers.

cmake_minimum_required(VERSION 3.10)

project(unicc VERSION 0.2.0 LANGUAGES C)

option(UNICC_BUILD_TESTS               "Build UniCCL unit tests (fake backends)" ON)
option(UNICC_BUILD_TESTS_INTEGRATION   "Build real-backend integration test (requires NCCL/RCCL at run time)" OFF)
option(UNICC_BUILD_EXAMPLES            "Build example programs" ON)

set(UNICC_SOURCES
    src/unicc_error.c
    src/unicc_loader.c
    src/unicc_vtable.c
    src/unicc_api.c
    src/backends/nccl.c
    src/backends/rccl.c
    src/backends/oneccl.c
    src/backends/eccl.c
    src/backends/unicc_bind.c)

if(WIN32)
    list(APPEND UNICC_SOURCES src/unicc_platform_windows.c)
else()
    list(APPEND UNICC_SOURCES src/unicc_platform_posix.c)
endif()

add_library(unicc ${UNICC_SOURCES})
add_library(unicc::unicc ALIAS unicc)

target_include_directories(unicc PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)
# Internal-only headers (e.g. unicc_native_id.h, shared with the fake
# fixtures) live under src/ and never get installed.
target_include_directories(unicc PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)

# C99 keeps the fake backends / tests simple and portable.
target_compile_features(unicc PUBLIC c_std_99)

# dl* lives in libdl on glibc < 2.34 (e.g. CentOS 8) and in libc on newer
# glibc; CMAKE_DL_LIBS carries the right link library on each.
target_link_libraries(unicc PUBLIC ${CMAKE_DL_LIBS})

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(unicc PRIVATE -Wall -Wextra)
endif()

# The static archive is frequently linked into downstream shared libraries
# (e.g. an upper-layer shared lib), so build it position-independent: without
# this, each consumer has to set POSITION_INDEPENDENT_CODE on the dependency
# itself (reported by the UniMemCom integration).
set_target_properties(unicc PROPERTIES POSITION_INDEPENDENT_CODE ON)

# --- install ----------------------------------------------------------------
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)

# Install the library and export it as the imported target unicc::unicc
# (install(EXPORT ... NAMESPACE unicc::) re-prefixes the built-in target name).
# The EXPORT must list the same public include dir the build interface uses -
# INTERFACE_INCLUDE_DIRECTORIES already carries $<INSTALL_INTERFACE:include>,
# so a consumer gets the install layout with no extra flag.
install(TARGETS unicc EXPORT uniccTargets
        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
        LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
        RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

# The public contract lives under include/ (docs: "include/ is the public
# contract"); the internal headers that the binders/fixtures share
# (unicc_bind.h, unicc_native_id.h) live under src/ and are never installed.
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES docs/API.md docs/BACKENDS.md docs/SUPPORT_MATRIX.md
        DESTINATION ${CMAKE_INSTALL_DOCDIR}/unicc)
install(FILES LICENSE
        DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/licenses/unicc)

# Export the target set into the package config dir, then generate a
# UniCCLConfig.cmake that a downstream find_package(UniCCL CONFIG) loads.
# The Config includes uniccTargets.cmake - generated by the install(EXPORT)
# below - so the pair always matches (unlike a Config that blindly includes a
# Targets file it never installs).
# Package config dir must match the find_package(UniCCL) name on
# case-sensitive filesystems: CMake searches <prefix>/(lib)/cmake/<Name>*/
# verbatim, so a lowercase "unicc" directory never matches.
set(UNICC_INSTALL_CMAKEDIR "${CMAKE_INSTALL_LIBDIR}/cmake/UniCCL")
install(EXPORT uniccTargets
        FILE uniccTargets.cmake
        NAMESPACE unicc::
        DESTINATION ${UNICC_INSTALL_CMAKEDIR})

set(UNICC_CMAKE_CONFIG_TEMPLATE "${CMAKE_CURRENT_SOURCE_DIR}/cmake/UniCCLConfig.cmake.in")
configure_package_config_file(
    "${UNICC_CMAKE_CONFIG_TEMPLATE}"
    "${CMAKE_CURRENT_BINARY_DIR}/UniCCLConfig.cmake"
    INSTALL_DESTINATION "${UNICC_INSTALL_CMAKEDIR}")

write_basic_package_version_file(
    "${CMAKE_CURRENT_BINARY_DIR}/UniCCLConfigVersion.cmake"
    VERSION "${PROJECT_VERSION}"
    COMPATIBILITY SameMajorVersion)

install(FILES
        "${CMAKE_CURRENT_BINARY_DIR}/UniCCLConfig.cmake"
        "${CMAKE_CURRENT_BINARY_DIR}/UniCCLConfigVersion.cmake"
        DESTINATION ${UNICC_INSTALL_CMAKEDIR})

# --- examples ---------------------------------------------------------------
if(UNICC_BUILD_EXAMPLES)
    add_executable(minimal examples/minimal.c)
    target_link_libraries(minimal PRIVATE unicc::unicc)
endif()

# --- tests ------------------------------------------------------------------
if(UNICC_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
